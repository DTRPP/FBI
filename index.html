    <script>
    let applications = [];
    let leaders = [];
    let logs = [];
    let currentLeader = null;
    let currentFilter = 'all';
    const OWNER_KEY = "FED-OWNER-2026"; // غيّر هذا الرمز إلى رمز سري خاص بك
    // character info temp storage between step 1 -> 2
    let pendingChar = {};
    function goTo(id){
    document.querySelectorAll('.screen').forEach(s=>s.classList.remove('active'));
    document.getElementById(id).classList.add('active');
    }
    async function loadAll(){
    try{ const r = await window.storage.get('fbi:applications', true); applications = r ? JSON.parse(r.value) : []; }catch(e){ applications = []; }
    try{ const r = await window.storage.get('fbi:leaders', true); leaders = r ? JSON.parse(r.value) : []; }catch(e){ leaders = []; }
    try{ const r = await window.storage.get('fbi:logs', true); logs = r ? JSON.parse(r.value) : []; }catch(e){ logs = []; }
    }
    async function saveApps(){ try{ await window.storage.set('fbi:applications', JSON.stringify(applications), true); }catch(e){} }
    async function saveLeaders(){ try{ await window.storage.set('fbi:leaders', JSON.stringify(leaders), true); }catch(e){} }
    async function saveLogs(){ try{ await window.storage.set('fbi:logs', JSON.stringify(logs), true); }catch(e){} }
    function esc(s){ return String(s ?? '').replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m])); }
    /* ---------- STEP 1 ---------- */
    function submitStep1(){
    const fullName = document.getElementById('c_fullName').value.trim();
    const dob = document.getElementById('c_dob').value;
    const idNumber = document.getElementById('c_idNumber').value.trim();
    const charAge = document.getElementById('c_charAge').value.trim();
    const realAge = document.getElementById('c_realAge').value;
    const experience = document.getElementById('c_experience').value.trim();
    const err = document.getElementById('c_err');
    if(!fullName || !dob || !idNumber || !charAge || !realAge || !experience){
    err.style.display = 'block'; return;
    }
    err.style.display = 'none';
    pendingChar = { fullName, dob, idNumber, charAge, realAge, experience };
    goTo('screen-app-form');
    }
    /* ---------- STEP 2 ---------- */
    async function submitApplication(){
    const motivation = document.getElementById('a_motivation').value.trim();
    const deptExp = document.getElementById('a_deptExp').value.trim();
    const availability = document.getElementById('a_availability').value.trim();
    const discord = document.getElementById('a_discord').value.trim();
    const notes = document.getElementById('a_notes').value.trim();
    const err = document.getElementById('a_err');
    if(!motivation || !availability || !discord){ err.style.display = 'block'; return; }
    err.style.display = 'none';
    const now = new Date();
    const app = {
    id: 'APP-' + Math.floor(1000 + Math.random()*9000),
    ...pendingChar,
    motivation, deptExp, availability, discord, notes,
    status: 'pending',
    submittedAt: now.toLocaleDateString('ar-EG'),
    ts: now.getTime()
    };
    applications.unshift(app);
    await saveApps();
    document.getElementById('refBox').textContent = 'الرقم المرجعي: ' + app.id;
    // reset forms
    ['c_fullName','c_idNumber','c_charAge','c_realAge','c_experience'].forEach(id=>document.getElementById(id).value='');
    document.getElementById('c_dob').value='';
    ['a_motivation','a_deptExp','a_availability','a_discord','a_notes'].forEach(id=>document.getElementById(id).value='');
    pendingChar = {};
    goTo('screen-confirmation');
    }
    /* ---------- OWNER GATE ---------- */
    function openOwnerGate(){ goTo('screen-owner-pass'); }
    function checkOwnerKey(){
    const val = document.getElementById('ownerKeyInput').value;
    const err = document.getElementById('owner_err');
    if(val === OWNER_KEY){
    err.style.display = 'none';
    document.getElementById('ownerKeyInput').value = '';
    renderLeaderMiniList();
    goTo('screen-owner-panel');
    } else {
    err.style.display = 'block';
    }
    }
    function renderLeaderMiniList(){
    const wrap = document.getElementById('leaderMiniList');
    if(!leaders.length){ wrap.innerHTML = '<div class="empty-note">No leaders registered yet.</div>'; return; }
    wrap.innerHTML = leaders.map(l=>`
    <div class="leader-mini">
    <div><div class="lm-name">${esc(l.name)}</div><div class="lm-meta">${esc(l.email)} · ${esc(l.rank)}</div></div>
    </div>
    `).join('');
    }
    async function saveLeader(){
    const email = document.getElementById('nl_email').value.trim().toLowerCase();
    const name = document.getElementById('nl_name').value.trim();
    const rank = document.getElementById('nl_rank').value;
    const password = document.getElementById('nl_password').value;
    const err = document.getElementById('nl_err');
    if(!email || !name || !password){ err.style.display='block'; return; }
    if(leaders.some(l=>l.email===email)){ err.textContent = 'This email is already registered.'; err.style.display='block'; return; }
    err.style.display = 'none';
    leaders.push({ email, name, rank, password, clearance:'FULL ACCESS' });
    await saveLeaders();
    ['nl_email','nl_name','nl_password'].forEach(id=>document.getElementById(id).value='');
    renderLeaderMiniList();
    }
    /* ---------- LEADER LOGIN ---------- */
    function leaderLogin(){
    const email = document.getElementById('l_email').value.trim().toLowerCase();
    const password = document.getElementById('l_password').value;
    const err = document.getElementById('l_err');
    const found = leaders.find(l=>l.email===email && l.password===password);
    if(!found){ err.style.display='block'; return; }
    err.style.display = 'none';
    currentLeader = found;
    document.getElementById('l_email').value=''; document.getElementById('l_password').value='';
    document.getElementById('dashName').textContent = found.name;
    document.getElementById('dashRank').textContent = found.rank;
    pushLog(`${found.name} signed in to the system`);
    renderApps(); renderLogs();
    goTo('screen-dashboard');
    }
    function leaderLogout(){
    if(currentLeader) pushLog(`${currentLeader.name} signed out of the system`);
    currentLeader = null;
    goTo('screen-landing');
    }
    /* ---------- DASHBOARD ---------- */
    function switchTab(tab){
    document.querySelectorAll('.tab-btn').forEach(b=>b.classList.toggle('active', b.dataset.tab===tab));
    document.querySelectorAll('.tab-panel').forEach(p=>p.classList.remove('active'));
    document.getElementById('tab-'+tab).classList.add('active');
    }
    function setFilter(f){
    currentFilter = f;
    document.querySelectorAll('.filter-btn').forEach(b=>b.classList.toggle('active', b.dataset.f===f));
    renderApps();
    }
    async function pushLog(text){
    logs.unshift({ text, ts: Date.now(), time: new Date().toLocaleString('en-US') });
    await saveLogs();
    renderLogs();
    }
    function renderApps(){
    const wrap = document.getElementById('appsList');
    const list = applications.filter(a => currentFilter==='all' ? true : a.status===currentFilter);
    if(!list.length){ wrap.innerHTML = '<div class="empty-note">No applications in this category.</div>'; return; }
    const statusLabel = {pending:'Pending', accepted:'Accepted', rejected:'Rejected'};
    wrap.innerHTML = list.map(a=>`
    <div class="app-card">
    <div class="app-top">
    <div><div class="name">${esc(a.fullName)}</div><div class="sub">${a.id} · Submitted on ${a.submittedAt}</div></div>
    <span class="status-pill ${a.status}">${statusLabel[a.status]}</span>
    </div>
    <div class="detail-grid">
    <div class="dg-item"><b>Date of Birth</b><span>${esc(a.dob)}</span></div>
    <div class="dg-item"><b>ID Number</b><span>${esc(a.idNumber)}</span></div>
    <div class="dg-item"><b>Character Age</b><span>${esc(a.charAge)}</span></div>
    <div class="dg-item"><b>Real Age</b><span>${esc(a.realAge)}</span></div>
    <div class="dg-item"><b>Availability</b><span>${esc(a.availability)}</span></div>
    <div class="dg-item"><b>Discord</b><span>${esc(a.discord)}</span></div>
    </div>
    <div class="long-field"><b>Experience</b>${esc(a.experience)}</div>
    <div class="long-field"><b>Motivation</b>${esc(a.motivation)}</div>
    ${a.deptExp ? `<div class="long-field"><b>Related Experience</b>${esc(a.deptExp)}</div>` : ''}
    ${a.notes ? `<div class="long-field"><b>Notes</b>${esc(a.notes)}</div>` : ''}
    ${a.status==='pending' ? `
    <div class="app-actions">
    <button class="btn-accept" onclick="decideApp('${a.id}','accepted')">Accept</button>
    <button class="btn-reject" onclick="decideApp('${a.id}','rejected')">Reject</button>
    </div>` : ''}
    </div>
    `).join('');
    }
    async function decideApp(id, status){
    const a = applications.find(a=>a.id===id);
    if(!a) return;
    a.status = status;
    await saveApps();
    pushLog(`${currentLeader.name} ${status==='accepted'?'accepted':'rejected'} the application of ${a.fullName} (${a.id})`);
    renderApps();
    }
    function renderLogs(){
    const wrap = document.getElementById('logsList');
    if(!logs.length){ wrap.innerHTML = '<div class="empty-note">No activity logged yet.</div>'; return; }
    wrap.innerHTML = logs.map(l=>`<div class="log-row"><span class="l-text">${esc(l.text)}</span><span class="l-time">${l.time}</span></div>`).join('');
    }
    loadAll();
